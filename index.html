        function handleFile(e, id) {

            const file = e.target.files[0];

            if (file) {

                const reader = new FileReader();

                reader.onload = (ev) => {

                    document.getElementById(`preview-${id}`).innerHTML = `<img src="${ev.target.result}" class="w-full h-full object-cover">`;

                };

                reader.readAsDataURL(file);

            }

        }
